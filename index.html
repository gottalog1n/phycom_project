Motion Tracking Radar

ผู้จัดทำ: 68070260 ดนุพล จิ้วพงศสวัสดิ์ | 68070249 ณัฐชนน แสงศรี | 68070324 อริยะ คุณรัตนโอฬาร | 68070298 พิสิษฐ์ ธิติเสรีวงศ์

📄 บทคัดย่อ (Abstract)

โครงงานนี้นำเสนอระบบเฝ้าระวังและตรวจจับสิ่งแปลกปลอมในพื้นที่ระยะไกล เพื่อแก้ปัญหาการพลาดการรับรู้ต่อสภาพแวดล้อมหรือเหตุการณ์ภายนอกขณะที่ผู้ใช้งานไม่อยู่ในบริเวณนั้น โดยระบบใช้เซนเซอร์วัดระยะทางอัลตราโซนิก (Ultrasonic Sensor) ตรวจจับความเคลื่อนไหวและระยะห่างของวัตถุอย่างต่อเนื่อง เมื่อพบวัตถุเข้ามาในรัศมีที่กำหนด (< 150 ซม.) ต่อเนื่องตามเกณฑ์ ระบบจะเข้าสู่โหมดเตือนภัยทันที โดยสั่งการให้ Buzzer ส่งสัญญาณเสียงเตือน, ไฟ LED สีแดงกะพริบ, หน้าจอ LCD I2C เปลี่ยนสถานะแสดงข้อความเตือนภัย และส่งข้อความแจ้งเตือนอัตโนมัติพร้อมระบุเวลาเข้าสู่ห้องแชท Discord ผ่าน Webhook Notification ช่วยให้ผู้ใช้สามารถรับรู้สถานการณ์ได้แบบเรียลไทม์ผ่านสมาร์ตโฟนหรือคอมพิวเตอร์อย่างแม่นยำและรวดเร็ว

⚙️ รายละเอียดระบบและหลักการทำงาน

1. สถาปัตยกรรมอินพุตและเอาต์พุต (I/O Architecture)

  • Input: Ultrasonic Sensor (HC-SR04) ตรวจวัดระยะห่างของวัตถุทุกๆ 60 ms แบบ Non-blocking
  • Processing Unit: ไมโครคอนโทรลเลอร์ประมวลผลคำนวณระยะทางและนับรอบการตรวจจับ (Consecutive Trigger Filter) เพื่อตัดปัญหา False Alarm
  • Local Output: จอ LCD I2C (16x2), Active Buzzer เตือนภัย และ LED Indicator (สีเขียว: ปกติ / สีแดง: แจ้งเตือน)
  • Remote Output: ระบบส่งสัญญาณแจ้งเตือนอัตโนมัติเข้าสู่ Discord Server ผ่าน Discord Webhook API

2. สถานะการทำงานของระบบ (System States)

  • Normal State (Area is Empty): ไฟ LED สีเขียวติดสว่าง, LED สีแดงและ Buzzer ปิด, หน้าจอ LCD แสดงข้อความ "Area is Empty"
  • Detection & Validation: เมื่อตรวจพบวัตถุระยะน้อยกว่า 150 ซม. ระบบจะนับสะสมรอบตรวจจับ (Trigger Count) หากตรวจพบต่อเนื่องครบ 5 ครั้ง จะเปลี่ยนสถานะเข้าสู่เตือนภัยทันที
  • Alert Mode (Warning & Notification): ไฟ LED สีเขียวดับ, LED สีแดงกะพริบพร้อม Buzzer ส่งสัญญาณเสียงต่อเนื่อง, หน้าจอ LCD แสดง "WARNING!! Foreign Body" พร้อมส่งสัญญาณเตือนเข้า Discord ทันที จนกว่าวัตถุจะออกจากพื้นที่

3. ตารางการเชื่อมต่อฮาร์ดแวร์ (Pin Connection)

อุปกรณ์ (Component) พินที่เชื่อมต่อ (Pin Assignment) หน้าที่ (Function)
Ultrasonic Sensor (HC-SR04) Trig: Pin 11, Echo: Pin 10 วัดระยะห่างวัตถุแบบสะท้อนคลื่นเสียง
LiquidCrystal I2C (16x2) SDA, SCL (Address 0x27) แสดงสถานะพื้นที่ Empty / Warning
Active Buzzer Pin 2 ส่งสัญญาณเสียงแจ้งเตือนเมื่อเกิดการบุกรุก
Red LED Pin 3 ไฟกะพริบเตือนภัยเมื่อมีวัตถุเข้ามาในระยะ
Green LED Pin 4 ไฟสถานะสแตนด์บายปกติเมื่อพื้นที่ปลอดภัย

// Motion Tracking Radar with LCD, Alert & Discord Notification
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// Ultrasonic sensor pins
const int trigPin = 11;
const int echoPin = 10;

// LED & Buzzer pins
const int redLed = 3;
const int greenLed = 4;
const int buzzer = 2;

// Initialize LCD 16x2 (I2C Address 0x27)
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Distance threshold & interval settings
const int alertDistance = 150;           // Alert if distance < 150 cm
const unsigned long checkInterval = 60;  // Loop measurement interval (ms)
unsigned long lastCheckTime = 0;

int triggerCount = 0;
enum RadarState { EMPTY, WARNING };
RadarState lastState = EMPTY;

void setup() {
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(redLed, OUTPUT);
  pinMode(greenLed, OUTPUT);
  pinMode(buzzer, OUTPUT);

  lcd.init();
  lcd.backlight();
  lcd.clear();

  lcdSetEmpty();
  setNormalState();
}

void loop() {
  unsigned long currentMillis = millis();

  if (currentMillis - lastCheckTime >= checkInterval) {
    lastCheckTime = currentMillis;

    float distance = getDistance();

    if (distance > 0 && distance < alertDistance) {
      triggerCount++;
    } else {
      triggerCount = 0;
    }

    // Trigger alert only after 5 consecutive detections
    if (triggerCount >= 5) {
      if (lastState != WARNING) {
        lastState = WARNING;
        lcdSetWarning();
        // Trigger Discord Webhook Notification via Serial / Network
      }
      alertMode();
      triggerCount = 0;
    } else {
      if (lastState != EMPTY) {
        lastState = EMPTY;
        lcdSetEmpty();
        setNormalState();
      }
    }
  }
}

float getDistance() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH, 30000);
  if (duration == 0) return 999.0;
  return (duration * 0.034 / 2);
}

void alertMode() {
  digitalWrite(greenLed, LOW);
  while (getDistance() < alertDistance) {
    digitalWrite(redLed, HIGH);
    digitalWrite(buzzer, HIGH);
    delay(100);
    digitalWrite(redLed, LOW);
    digitalWrite(buzzer, LOW);
    delay(100);
  }
  lcdSetEmpty();
  setNormalState();
  lastState = EMPTY;
}

void setNormalState() {
  digitalWrite(greenLed, HIGH);
  digitalWrite(redLed, LOW);
  digitalWrite(buzzer, LOW);
}

void lcdSetEmpty() {
  lcd.clear();
  lcd.setCursor(1, 0);
  lcd.print("Area is Empty");
  lcd.setCursor(0, 1);
  lcd.print("                ");
}

void lcdSetWarning() {
  lcd.clear();
  lcd.setCursor(4, 0);
  lcd.print("WARNING!!");
  lcd.setCursor(2, 1);
  lcd.print("Foreign Body");
}
