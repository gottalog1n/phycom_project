Motion Tracking Radar

📄 บทคัดย่อ (Abstract)

โครงงานนี้นำเสนอระบบเฝ้าระวังพื้นที่ระยะไกลเพื่อแก้ปัญหาการพลาดการรับรู้ต่อสิ่งแวดล้อมหรือเหตุการณ์ภายนอกขณะที่ไม่อยู่ในบริเวณนั้น โดยระบบใช้เซนเซอร์วัดระยะทางอัลตราโซนิก (Ultrasonic Sensor) ทำหน้าที่ตรวจจับการเคลื่อนไหวหรือการบุกรุกพื้นที่ เมื่อพบวัตถุเข้ามาในระยะที่กำหนด ไมโครคอนโทรลเลอร์จะประมวลผลและส่งสัญญาณแจ้งเตือนแบบครอบคลุม ทั้งในพื้นที่จริงผ่านเสียงสัญญาณเตือน (Buzzer) และไฟแสดงสถานะ (LED) รวมถึงส่งการแจ้งเตือนทันทีไปยังผู้ใช้ผ่าน LINE Notification ช่วยให้รับรู้เหตุการณ์ได้อย่างรวดเร็วและเพิ่มความปลอดภัยให้กับพื้นที่แม้ไม่ได้อยู่ในจุดเกิดเหตุ

⚙️ รายละเอียดระบบและหลักการทำงาน

  • Input: ระยะห่างจากเซนเซอร์อัลตราโซนิก (HC-SR04), ปุ่มกด Reset/Setup
  • Processing: ไมโครคอนโทรลเลอร์ ESP32 ตรวจจับเงื่อนไขระยะห่าง
  • Output: สัญญาณเตือน Buzzer และหลอดไฟ LED แสดงสถานะการเคลื่อนไหว

🎬 คลิปวิดีโอสาธิตการทำงาน

วิดีโอนี้อธิบายการทำงานของระบบ แสดงตัวชิ้นงานฮาร์ดแวร์จริง และเห็นหน้าสมาชิกทุกคนในกลุ่มครบถ้วน

const int echoPin = 10;

const int redLed = 3;
const int greenLed = 4;

// Initialize the LCD (16x2). Check the I2C address (commonly 0x27 or 0x3F)
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Settings
const int alertDistance = 150;           // Alarm triggers if distance is less than 150cm
const unsigned long checkInterval = 60;  // Measurement interval (ms)
unsigned long lastCheckTime = 0;

// Variable to count consecutive detections under threshold
int triggerCount = 0;

// Enum to record state (for updating LCD only when state changes)
enum RadarState { EMPTY, WARNING };
RadarState lastState = EMPTY;

void setup() {
  // Initialize ultrasonic sensor pins
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

  // Initialize LED and buzzer pins
  pinMode(redLed, OUTPUT);
  pinMode(greenLed, OUTPUT);
  pinMode(buzzer, OUTPUT);

  // Initialize LCD
  lcd.init();
  lcd.backlight();
  lcd.clear();

  // Set initial status: LCD shows "Area is Empty", green LED on
  lcdSetEmpty();
  setNormalState();
